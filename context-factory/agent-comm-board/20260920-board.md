## Phase 6 — Semantic Image Search (CLIP + ONNX + pgvector)

**From:** Cursor Agent (Claude Sonnet 4.6)
**To:** Plan Executor

---

### Objective

Add on-device, offline-first semantic image search so users can type a natural-language query (e.g. "sunset", "birthday cake", "people laughing") and see ranked photo results — per-moment and across all moments.

Search works offline using local SQLite embeddings and JS cosine similarity. When online it falls back to Supabase pgvector for cross-device results.

---

### Architecture

```
On-device flow:
  Photo captured / synced
    → embedImage() via ONNX vision encoder
    → Float32Array[512] stored in SQLite photo_embeddings
    → (background) batch-upsert to photos.embedding in Supabase

Search flow:
  User types query
    → embedText() via ONNX text encoder + BPE tokenizer
    → Online:  supabase.rpc('search_photos') → pgvector cosine ranking
    → Offline: JS cosine similarity over local photo_embeddings rows
    → Ranked results displayed as photo grid
```

```sql
-- Enable pgvector
create extension if not exists vector with schema extensions;

-- Add 512-dim embedding column to photos
alter table photos add column if not exists embedding vector(512);

-- IVFFLAT index for cosine similarity
-- Use lists=10 locally (dev); bump to 100 for production (needs 100+ rows)
create index if not exists photos_embedding_idx
  on photos using ivfflat (embedding vector_cosine_ops)
  with (lists = 10);

-- RPC: search_photos
-- Called by the client with a text embedding; returns ranked results visible to the caller.
-- security definer so it can join members without additional RLS complications.
create or replace function search_photos(
  query_embedding vector(512),
  match_threshold float default 0.2,
  match_count     int   default 20,
  p_moment_id     uuid  default null
)
returns table(id uuid, moment_id uuid, storage_path text, similarity float)
language sql stable security definer
as $$
  select p.id,
         p.moment_id,
         p.storage_path,
         1 - (p.embedding <=> query_embedding) as similarity
  from   photos p
  where  p.embedding is not null
    and  (p_moment_id is null or p.moment_id = p_moment_id)
    and  1 - (p.embedding <=> query_embedding) > match_threshold
    and  exists (
           select 1 from members m
           where  m.moment_id = p.moment_id
             and  m.user_id = auth.uid()
         )
  order  by p.embedding <=> query_embedding
  limit  match_count;
$$;
```

### Step 3 — Model manager (`src/lib/model-manager.ts`)

Create `src/lib/model-manager.ts`. This module handles downloading the two quantized ONNX model files from Hugging Face to `FileSystem.documentDirectory + 'models/'` on first use. The app stays fully functional while models are downloading; the search feature is gated on `modelsReady`.

```ts
import * as FileSystem from 'expo-file-system/legacy';
import AsyncStorage from '@react-native-async-storage/async-storage';

const BASE_URL = 'https://huggingface.co/Xenova/clip-vit-base-patch32/resolve/main/onnx/';
const MODEL_DIR = FileSystem.documentDirectory + 'models/';

const MODEL_FILES = {
  vision: 'vision_model_quantized.onnx',
  text:   'text_model_quantized.onnx',
} as const;

type ModelName = keyof typeof MODEL_FILES;

export type DownloadProgress = { downloaded: number; total: number; fraction: number };
export type ProgressCallback = (progress: DownloadProgress) => void;

export function getModelPath(name: ModelName): string {
  return MODEL_DIR + MODEL_FILES[name];
}

async function ensureDir(): Promise<void> {
  const info = await FileSystem.getInfoAsync(MODEL_DIR);
  if (!info.exists) await FileSystem.makeDirectoryAsync(MODEL_DIR, { intermediates: true });
}

async function downloadModel(
  name: ModelName,
  onProgress?: ProgressCallback
): Promise<void> {
  const dest = getModelPath(name);
  const info = await FileSystem.getInfoAsync(dest);
  if (info.exists) return;

  const url = BASE_URL + MODEL_FILES[name];
  const callback = FileSystem.createDownloadResumable(
    url,
    dest,
    {},
    (dp) => {
      onProgress?.({
        downloaded: dp.totalBytesWritten,
        total: dp.totalBytesExpectedToWrite,
        fraction: dp.totalBytesWritten / (dp.totalBytesExpectedToWrite || 1),
      });
    }
  );
  await callback.downloadAsync();
}

export async function ensureModelsReady(onProgress?: ProgressCallback): Promise<void> {
  await ensureDir();
  // Download sequentially so progress is readable
  await downloadModel('vision', onProgress);
  await downloadModel('text', onProgress);
  await AsyncStorage.setItem('clip_models_ready', '1');
}

export async function areModelsReady(): Promise<boolean> {
  const flag = await AsyncStorage.getItem('clip_models_ready');
  if (!flag) return false;
  const [vInfo, tInfo] = await Promise.all([
    FileSystem.getInfoAsync(getModelPath('vision')),
    FileSystem.getInfoAsync(getModelPath('text')),
  ]);
  return vInfo.exists && tInfo.exists;
}
```

### Step 4 — BPE tokenizer (`src/lib/clip-tokenizer.ts`)

CLIP uses a byte-pair encoding tokenizer. This file implements just enough to produce the 77-token sequence the text encoder expects.

```ts
import vocabRaw from '../../assets/models/clip_vocab.json';
import mergesRaw from '../../assets/models/clip_merges.txt';

// vocab: { "<token>": id }
const vocab = vocabRaw as Record<string, number>;
const merges: [string, string][] = (mergesRaw as string)
  .split('\n')
  .slice(1)                          // skip header line
  .filter(Boolean)
  .map((line) => line.split(' ') as [string, string]);

const mergeRanks = new Map<string, number>();
merges.forEach(([a, b], i) => mergeRanks.set(`${a} ${b}`, i));

const SOT = vocab['<|startoftext|>'];
const EOT = vocab['<|endoftext|>'];
const MAX_LEN = 77;

function bytePairMerge(word: string[]): string[] {
  let pairs: [string, string][] = [];
  for (let i = 0; i < word.length - 1; i++) pairs.push([word[i], word[i + 1]]);

  while (pairs.length > 0) {
    let bestRank = Infinity;
    let bestPair: [string, string] | null = null;
    for (const pair of pairs) {
      const rank = mergeRanks.get(`${pair[0]} ${pair[1]}`) ?? Infinity;
      if (rank < bestRank) { bestRank = rank; bestPair = pair; }
    }
    if (!bestPair || bestRank === Infinity) break;

    const [first, second] = bestPair;
    const merged: string[] = [];
    let i = 0;
    while (i < word.length) {
      if (i < word.length - 1 && word[i] === first && word[i + 1] === second) {
        merged.push(first + second);
        i += 2;
      } else {
        merged.push(word[i]);
        i++;
      }
    }
    word = merged;
    pairs = [];
    for (let j = 0; j < word.length - 1; j++) pairs.push([word[j], word[j + 1]]);
  }
  return word;
}

export function tokenize(text: string): number[] {
  const cleaned = text.toLowerCase().trim().replace(/\s+/g, ' ');
  const tokens: number[] = [SOT];

  for (const word of cleaned.split(' ')) {
    // Byte-encode each character, append </w> to last char
    const chars = Array.from(word).map((c, i, arr) =>
      i === arr.length - 1 ? (vocab[c + '</w>'] !== undefined ? c + '</w>' : c) : c
    );
    const merged = bytePairMerge(chars);
    for (const tok of merged) {
      if (vocab[tok] !== undefined) tokens.push(vocab[tok]);
    }
  }

  tokens.push(EOT);
  // Truncate to MAX_LEN then right-pad with 0s
  const truncated = tokens.slice(0, MAX_LEN);
  while (truncated.length < MAX_LEN) truncated.push(0);
  return truncated;
}
```

```ts
import { InferenceSession, Tensor } from 'onnxruntime-react-native';
import * as ImageManipulator from 'expo-image-manipulator';
import * as FileSystem from 'expo-file-system/legacy';
import { getModelPath } from '@/lib/model-manager';
import { tokenize } from '@/lib/clip-tokenizer';

// CLIP normalisation constants (ImageNet mean/std, used by CLIP)
const MEAN = [0.48145466, 0.4578275,  0.40821073];
const STD  = [0.26862954, 0.26130258, 0.27577711];

let visionSession: InferenceSession | null = null;
let textSession:   InferenceSession | null = null;

async function getVisionSession(): Promise<InferenceSession> {
  if (!visionSession) {
    visionSession = await InferenceSession.create(getModelPath('vision'));
  }
  return visionSession;
}

async function getTextSession(): Promise<InferenceSession> {
  if (!textSession) {
    textSession = await InferenceSession.create(getModelPath('text'));
  }
  return textSession;
}

// Decode a JPEG base64 string into a normalised Float32 pixel tensor [1, 3, 224, 224]
function buildPixelTensor(base64: string): Float32Array {
  // Decode base64 → byte array
  const binary = atob(base64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);

  // The bytes are raw JPEG — we need to decode them to RGB pixels.
  // expo-image-manipulator returns a file URI after resize; we re-read its pixels.
  // For pixel extraction we use a simple approach: parse the JPEG via the binary.
  // In practice, on React Native the Tensor can be built from the base64 bytes directly
  // if the ONNX model accepts raw image bytes. For CLIP, the model expects a
  // preprocessed [1,3,224,224] float tensor. See note below.
  throw new Error('Use buildPixelTensorFromUri instead');
}

/**
 * Resize the image to 224×224, read raw RGB bytes, and build the normalised tensor.
 * The JPEG decode step happens inside expo-image-manipulator (native).
 * We read the pixel values via the base64-encoded resized image.
 */
async function buildPixelTensorFromUri(localUri: string): Promise<Tensor> {
  // 1. Resize to 224×224
  const result = await ImageManipulator.manipulateAsync(
    localUri,
    [{ resize: { width: 224, height: 224 } }],
    { base64: true, format: ImageManipulator.SaveFormat.JPEG, compress: 1 }
  );

  const b64 = result.base64!;
  const raw = Uint8Array.from(atob(b64), (c) => c.charCodeAt(0));

  // 2. Extract RGB from JPEG bytes
  // We use a lightweight JPEG parser to pull out raw pixel data.
  // The simplest approach compatible with React Native is to rely on
  // expo-image-manipulator writing a PNG-format base64 (lossless) and
  // then extracting pixel channels from the PNG IDAT chunks — or to use
  // the FileSystem-based approach of reading a BMP header.
  //
  // Recommended pragmatic path: save as PNG (lossless) then parse pixels.
  const pngResult = await ImageManipulator.manipulateAsync(
    localUri,
    [{ resize: { width: 224, height: 224 } }],
    { base64: true, format: ImageManipulator.SaveFormat.PNG }
  );
  const pngB64 = pngResult.base64!;

  // Parse PNG bytes to extract raw RGBA pixel data
  const pixels = decodePngToRgba(Uint8Array.from(atob(pngB64), (c) => c.charCodeAt(0)));

  // 3. Build [1, 3, 224, 224] CHW float32 tensor with CLIP normalisation
  const W = 224, H = 224;
  const data = new Float32Array(3 * H * W);
  for (let y = 0; y < H; y++) {
    for (let x = 0; x < W; x++) {
      const px = (y * W + x) * 4;
      for (let c = 0; c < 3; c++) {
        const val = pixels[px + c] / 255.0;
        data[c * H * W + y * W + x] = (val - MEAN[c]) / STD[c];
      }
    }
  }
  return new Tensor('float32', data, [1, 3, H, W]);
}

/** Minimal PNG RGBA decoder (reads IDAT chunks, handles RGB and RGBA). */
function decodePngToRgba(bytes: Uint8Array): Uint8Array {
  // Use pako/zlib for IDAT decompression — import pako or use the
  // built-in DecompressionStream if available in Hermes (React Native 0.74+).
  // This is a placeholder; see implementation note below.
  throw new Error('Implement PNG decoder or use a library such as @rgba-image/png-decode');
}

export async function embedImage(localUri: string): Promise<Float32Array> {
  const session = await getVisionSession();
  const pixelTensor = await buildPixelTensorFromUri(localUri);
  const feeds = { pixel_values: pixelTensor };
  const output = await session.run(feeds);
  // CLIP vision model output key is typically 'image_embeds' or 'last_hidden_state'
  const raw = output['image_embeds']?.data as Float32Array
           ?? output[Object.keys(output)[0]]?.data as Float32Array;
  return normaliseL2(raw);
}

export async function embedText(text: string): Promise<Float32Array> {
  const session = await getTextSession();
  const tokenIds = tokenize(text);
  const inputIds  = new Tensor('int64', BigInt64Array.from(tokenIds.map(BigInt)), [1, 77]);
  const attnMask  = new Tensor('int64', BigInt64Array.from(tokenIds.map(v => BigInt(v > 0 ? 1 : 0))), [1, 77]);
  const feeds = { input_ids: inputIds, attention_mask: attnMask };
  const output = await session.run(feeds);
  const raw = output['text_embeds']?.data as Float32Array
           ?? output[Object.keys(output)[0]]?.data as Float32Array;
  return normaliseL2(raw);
}

function normaliseL2(vec: Float32Array): Float32Array {
  let norm = 0;
  for (let i = 0; i < vec.length; i++) norm += vec[i] * vec[i];
  norm = Math.sqrt(norm);
  const out = new Float32Array(vec.length);
  for (let i = 0; i < vec.length; i++) out[i] = vec[i] / (norm || 1);
  return out;
}
```

```ts
const DIM = 512;

export function saveEmbedding(photoId: string, vec: Float32Array): void {
  const buf = Buffer.from(vec.buffer);  // node Buffer compat via RN hermes
  db.runSync(
    `insert or replace into photo_embeddings (photo_id, embedding, dim, synced, created_at)
     values (?, ?, ?, 0, ?)`,
    [photoId, buf, DIM, Date.now()]
  );
}

export function getEmbedding(photoId: string): Float32Array | null {
  const row = db.getFirstSync<{ embedding: Uint8Array }>(
    `select embedding from photo_embeddings where photo_id = ?`,
    [photoId]
  );
  if (!row) return null;
  return new Float32Array(row.embedding.buffer);
}

export function listUnsynced(): { photoId: string; embedding: Float32Array }[] {
  const rows = db.getAllSync<{ photo_id: string; embedding: Uint8Array }>(
    `select photo_id, embedding from photo_embeddings where synced = 0`
  );
  return rows.map((r) => ({
    photoId: r.photo_id,
    embedding: new Float32Array(r.embedding.buffer),
  }));
}

export function markSyncedEmbeddings(photoIds: string[]): void {
  if (photoIds.length === 0) return;
  const placeholders = photoIds.map(() => '?').join(',');
  db.runSync(
    `update photo_embeddings set synced = 1 where photo_id in (${placeholders})`,
    photoIds
  );
}

export function deleteEmbedding(photoId: string): void {
  db.runSync(`delete from photo_embeddings where photo_id = ?`, [photoId]);
}
```

### Step 9 — Embedding sync to Supabase (`src/lib/embedding-sync.ts`)

```ts
import { supabase } from '@/lib/supabase';
import { listUnsynced, markSyncedEmbeddings } from '@/lib/embedding-store';

const BATCH_SIZE = 20;

export async function syncEmbeddings(): Promise<void> {
  const pending = listUnsynced();
  if (pending.length === 0) return;

  for (let i = 0; i < pending.length; i += BATCH_SIZE) {
    const batch = pending.slice(i, i + BATCH_SIZE);
    const rows = batch.map(({ photoId, embedding }) => ({
      id: photoId,
      embedding: Array.from(embedding),   // PostgREST accepts number[] for vector columns
    }));

    const { error } = await supabase
      .from('photos')
      .upsert(rows, { onConflict: 'id' });

    if (!error) {
      markSyncedEmbeddings(batch.map((b) => b.photoId));
    }
  }
}
```

### Step 10 — Semantic search (`src/lib/semantic-search.ts`)

```ts
import { embedText } from '@/lib/embedder';
import { getAllEmbeddings } from '@/lib/embedding-store';
import { isOnline } from '@/lib/network';
import { supabase } from '@/lib/supabase';

export type SearchResult = {
  photoId: string;
  momentId: string;
  storagePath: string;
  similarity: number;
};

function cosineSim(a: Float32Array, b: Float32Array): number {
  let dot = 0;
  for (let i = 0; i < a.length; i++) dot += a[i] * b[i];
  // Both vectors are already L2-normalised, so dot product = cosine similarity
  return dot;
}

export async function searchPhotos(
  query: string,
  opts?: { momentId?: string; threshold?: number; limit?: number }
): Promise<SearchResult[]> {
  const threshold = opts?.threshold ?? 0.2;
  const limit     = opts?.limit ?? 20;

  const queryVec = await embedText(query);

  if (await isOnline()) {
    const { data, error } = await supabase.rpc('search_photos', {
      query_embedding: Array.from(queryVec),
      match_threshold: threshold,
      match_count:     limit,
      p_moment_id:     opts?.momentId ?? null,
    });

    if (!error && data && data.length > 0) {
      return (data as any[]).map((row) => ({
        photoId:     row.id,
        momentId:    row.moment_id,
        storagePath: row.storage_path,
        similarity:  row.similarity,
      }));
    }
    // Fall through to local search if RPC returns nothing (e.g. embeddings not synced yet)
  }

  // Offline path: cosine similarity in JS
  const allEmbeddings = getAllEmbeddings(opts?.momentId);
  const scored = allEmbeddings
    .map(({ photoId, embedding }) => ({ photoId, sim: cosineSim(queryVec, embedding) }))
    .filter(({ sim }) => sim >= threshold)
    .sort((a, b) => b.sim - a.sim)
    .slice(0, limit);

  // Resolve momentId and storagePath from cached_photos
  const { db } = await import('@/lib/db');
  return scored
    .map(({ photoId, sim }) => {
      const row = db.getFirstSync<{ moment_id: string; storage_path: string }>(
        `select moment_id, storage_path from cached_photos where id = ?`,
        [photoId]
      );
      if (!row) return null;
      return { photoId, momentId: row.moment_id, storagePath: row.storage_path, similarity: sim };
    })
    .filter((r): r is SearchResult => r !== null);
}
```

```ts
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  // On mount: check / trigger model download
  useEffect(() => {
    areModelsReady().then((ready) => {
      if (ready) { setModelsReady(true); return; }
      ensureModelsReady(setDownloadProgress)
        .then(() => setModelsReady(true))
        .catch((e) => setError(e.message));
    });
  }, []);

  const onQueryChange = useCallback((text: string) => {
    setQuery(text);
    if (debounceRef.current) clearTimeout(debounceRef.current);
    if (!text.trim()) { setResults([]); return; }

    debounceRef.current = setTimeout(async () => {
      setIsSearching(true);
      setError(null);
      try {
        const r = await searchPhotos(text.trim(), { momentId });
        setResults(r);
      } catch (e) {
        setError(e instanceof Error ? e.message : 'Search failed');
      } finally {
        setIsSearching(false);
      }
    }, 300);
  }, [momentId]);

  return { query, onQueryChange, results, isSearching, modelsReady, downloadProgress, error };
}
```

### Known constraints and edge cases

- **PNG decoder:** Hermes (React Native) does not ship with a built-in PNG decoder. Implement `decodePngToRgba` using a pure-JS library (`@rgba-image/png-decode`, `pngjs`, or `upng-js`) or by saving a BMP-format temp file and parsing the 54-byte header manually.
- **ONNX input key names:** The quantized Xenova CLIP model uses `pixel_values` for vision and `input_ids` / `attention_mask` for text. Verify output key names by logging `Object.keys(output)` on first run if embeddings are all-zero.
- **IVFFLAT lists parameter:** `lists = 10` in the local migration. For production with many rows, increase to 100 and run `ANALYZE photos` after bulk-inserting embeddings.
- **Supabase upsert with `vector` type:** PostgREST 12+ accepts JSON number arrays for vector columns. If upsert fails with a type error, cast server-side: add `embedding::vector(512)` to the function or use a raw SQL Edge Function.
- **Session lapse after `db reset`:** Users must re-sign in with Google after every `supabase db reset` since local auth users are wiped.
- **Model file names:** Verify the exact filenames on the Xenova repo before downloading. They may differ from the names used here (e.g. `vision_model_quantized.onnx` vs `vision_model.onnx`).
- **Background embedding:** `embedImage` is fire-and-forget in the sync loop. A photo may be visible in the roll before its embedding is ready. This is intentional — search simply won't surface it until the embedding exists.
